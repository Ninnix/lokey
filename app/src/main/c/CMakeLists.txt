cmake_minimum_required(VERSION 3.22)
project(lokey LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LOKEY_HEXAGON "Build the optional Hexagon backend" OFF)
set(GGML_HEXAGON ${LOKEY_HEXAGON} CACHE BOOL "" FORCE)
set(LOKEY_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." CACHE PATH "Parent of the three Loina forks")
foreach(option LOINA_DIFFUSION LOINA_CLI LOINA_TESTS LOINA_SERVER LOINA_BACKEND_DL GGML_VULKAN GGML_OPENCL GGML_OPENMP)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LOINA_LLAMA OFF CACHE BOOL "" FORCE)
set(LOINA_SPEECH ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LOINA_RUNTIME_ROOT "${LOKEY_RUNTIME_ROOT}" CACHE PATH "" FORCE)
add_subdirectory("${LOKEY_RUNTIME_ROOT}/loina-core" loina)
add_library(lokey SHARED lokey.c)
target_include_directories(lokey PRIVATE "${LOKEY_RUNTIME_ROOT}/llama.cpp/include" "${LOKEY_RUNTIME_ROOT}/whisper.cpp/include")
target_link_libraries(lokey PRIVATE loina-core llama parakeet mtmd log m)
target_link_options(lokey PRIVATE "-Wl,-z,max-page-size=16384")
if(LOKEY_HEXAGON)
    target_compile_definitions(lokey PRIVATE LOKEY_HEXAGON)
    if(LOKEY_HTP_LIB_DIR)
        file(GLOB htp_skels "${LOKEY_HTP_LIB_DIR}/libggml-htp-v*.so")
        if(NOT htp_skels)
            message(FATAL_ERROR "LOKEY_HTP_LIB_DIR has no Hexagon kernels")
        endif()
    else()
        foreach(version v73 v75 v79 v81 v85)
            list(APPEND htp_skels "${CMAKE_CURRENT_BINARY_DIR}/loina/runtimes/llama/ggml/src/ggml-hexagon/libggml-htp-${version}.so")
        endforeach()
    endif()
    add_custom_target(lokey-htp
        COMMAND ${CMAKE_COMMAND} -E make_directory "${LOKEY_HTP_OUTPUT}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${htp_skels} "${LOKEY_HTP_OUTPUT}"
        VERBATIM)
    if(NOT LOKEY_HTP_LIB_DIR)
        add_dependencies(lokey-htp htp-v73 htp-v75 htp-v79 htp-v81 htp-v85)
    endif()
    add_dependencies(lokey lokey-htp)
    target_link_libraries(lokey PRIVATE dl)
endif()
